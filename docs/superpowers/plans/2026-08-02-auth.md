# Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver functional Supabase email/password Login + Register screens with route protection, wired to activate when credentials are provided, while remaining fully runnable now (no live backend).

**Architecture:** Uses the official `@supabase/ssr` client/server pattern. A root `src/proxy.ts` (Next 16 — `middleware` was renamed to `proxy`) guards `/(app)`, but only once `NEXT_PUBLIC_SUPABASE_URL` + `NEXT_PUBLIC_SUPABASE_ANON_KEY` exist; without them every route passes through for continued development. Forms use react-hook-form + zod. The Supabase client library, database types, schema migration, dependencies, and shadcn UI components **already exist in the working tree but are uncommitted** — this plan commits that baseline, then authors the missing auth screens.

**Tech Stack:** Next.js 16 App Router, TypeScript, Tailwind v4 (shadcn/radix-nova tokens), `@supabase/ssr` + `@supabase/supabase-js`, `react-hook-form` + `zod` + `@hookform/resolvers`. No unit-test framework is configured; verification is `npm run build` + `npm run lint` + browser dev-mode inspection.

**Verification convention (this project):** build + lint + manual browser checks. Supabase login/signup cannot be exercised without real credentials (deferred by design) — dev-mode behavior and compile correctness are the acceptance bar here.

### Task 3: Login form (`src/app/(auth)/login/page.tsx`)

Replace the stub with a real form: react-hook-form + zod validation, Supabase `signInWithPassword`, inline errors, redirect to `/dashboard`. When Supabase is unconfigured, render a visible notice and disable submit (no crash).

**Files:**
- Overwrite: `src/app/(auth)/login/page.tsx`

- [ ] **Step 1: Write the page**

Overwrite `src/app/(auth)/login/page.tsx`:
```tsx
"use client";

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useRouter } from "next/navigation";
import { useState } from "react";
import { z } from "zod";

import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { createClient } from "@/lib/supabase/client";
import { isSupabaseConfigured } from "@/lib/supabase/config";

const schema = z.object({
  email: z.string().email("Enter a valid email."),
  password: z.string().min(1, "Password is required."),
});
type FormValues = z.infer<typeof schema>;

export default function LoginPage() {
  const router = useRouter();
  const configured = isSupabaseConfigured();
  const [formError, setFormError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<FormValues>({ resolver: zodResolver(schema) });

  async function onSubmit(values: FormValues) {
    setFormError(null);
    if (!configured) return;

    const supabase = createClient();
    const { error } = await supabase.auth.signInWithPassword(values);

    if (error) {
      setFormError(
        error.code === "invalid_credentials"
          ? "Invalid email or password."
          : error.message
      );
      return;
    }

    router.push("/dashboard");
    router.refresh();
  }

  return (
    <div className="space-y-4 rounded-lg border border-default bg-surface p-6">
      <h1 className="text-lg font-semibold">Log in</h1>
      <p className="text-sm text-faint">Welcome back to EngineerOS.</p>

      {!configured && (
        <p className="rounded-md border border-dashed border-default bg-muted p-2 text-xs text-faint">
          Supabase is not configured. Add NEXT_PUBLIC_SUPABASE_URL and
          NEXT_PUBLIC_SUPABASE_ANON_KEY to <code>.env.local</code>, then restart
          the dev server to enable login.
        </p>
      )}

      <form onSubmit={handleSubmit(onSubmit)} className="space-y-4" noValidate>
        <div className="space-y-2">
          <Label htmlFor="email">Email</Label>
          <Input id="email" type="email" autoComplete="email" {...register("email")} />
          {errors.email && <p className="text-xs text-destructive">{errors.email.message}</p>}
        </div>

        <div className="space-y-2">
          <Label htmlFor="password">Password</Label>
          <Input
            id="password"
            type="password"
            autoComplete="current-password"
            {...register("password")}
          />
          {errors.password && (
            <p className="text-xs text-destructive">{errors.password.message}</p>
          )}
        </div>

        {formError && <p className="text-sm text-destructive">{formError}</p>}

        <Button type="submit" className="w-full" disabled={!configured || isSubmitting}>
          {isSubmitting ? "Logging in…" : "Log in"}
        </Button>
      </form>
    </div>
  );
}
```

```tsx
"use client";

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useRouter } from "next/navigation";
import { useState } from "react";
import { z } from "zod";

import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { createClient } from "@/lib/supabase/client";
import { isSupabaseConfigured } from "@/lib/supabase/config";

const schema = z.object({
  email: z.string().email("Enter a valid email."),
  password: z.string().min(8, "At least 8 characters."),
});
type FormValues = z.infer<typeof schema>;

export default function RegisterPage() {
  const router = useRouter();
  const configured = isSupabaseConfigured();
  const [formError, setFormError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<FormValues>({ resolver: zodResolver(schema) });

  async function onSubmit(values: FormValues) {
    setFormError(null);
    if (!configured) return;

    const supabase = createClient();
    const { error } = await supabase.auth.signUp(values);

    if (error) {
      setFormError(
        error.code === "already_registered" || error.code === "user_already_exists"
          ? "An account with that email already exists."
          : error.message
      );
      return;
    }

    router.push("/dashboard");
    router.refresh();
  }

  return (
    <div className="space-y-4 rounded-lg border border-default bg-surface p-6">
      <h1 className="text-lg font-semibold">Create your account</h1>
      <p className="text-sm text-faint">A workspace is created for you automatically.</p>

      {!configured && (
        <p className="rounded-md border border-dashed border-default bg-muted p-2 text-xs text-faint">
          Supabase is not configured. Add NEXT_PUBLIC_SUPABASE_URL and
          NEXT_PUBLIC_SUPABASE_ANON_KEY to <code>.env.local</code> to enable signup.
        </p>
      )}

      <form onSubmit={handleSubmit(onSubmit)} className="space-y-4" noValidate>
        <div className="space-y-2">
          <Label htmlFor="email">Email</Label>
          <Input id="email" type="email" autoComplete="email" {...register("email")} />
          {errors.email && <p className="text-xs text-destructive">{errors.email.message}</p>}
        </div>

        <div className="space-y-2">
          <Label htmlFor="password">Password</Label>
          <Input
            id="password"
            type="password"
            autoComplete="new-password"
            {...register("password")}
          />
          {errors.password && (
            <p className="text-xs text-destructive">{errors.password.message}</p>
          )}
        </div>

        {formError && <p className="text-sm text-destructive">{formError}</p>}

        <Button type="submit" className="w-full" disabled={!configured || isSubmitting}>
          {isSubmitting ? "Creating account…" : "Create account"}
        </Button>
      </form>
    </div>
  );
}
```
